use bevy::prelude::*;
use bevy_immediate::{
    BevyImmediatePlugin, ImmCtx,
    ui::{CapsUi, clicked::ImmUiClicked, text::ImmUiText},
};
use bevy_immediate_tutorial::{app, button, camera, panel};

#[derive(Resource, Default)]
struct Counter {
    value: u32,
}

fn main() {
    app("04 - Counter")
        .init_resource::<Counter>()
        .add_plugins(BevyImmediatePlugin::<CapsUi>::new())
        .add_systems(Startup, camera)
        .add_systems(Update, draw_ui)
        .run();
}

fn draw_ui(ctx: ImmCtx<CapsUi>, mut counter: ResMut<Counter>) {
    let mut ui = ctx.build_immediate_root("lesson04");
    ui.ch_id("panel").on_spawn_insert(panel).add(|ui| {
        let mut add = ui.ch_id("add").on_spawn_insert(button).add(|ui| {
            ui.ch_id("caption").text("+1");
        });
        if add.clicked() {
            counter.value += 1;
        }

        let mut reset = ui.ch_id("reset").on_spawn_insert(button).add(|ui| {
            ui.ch_id("caption").text("Reset");
        });
        if reset.clicked() {
            counter.value = 0;
        }

        // 先处理输入，再描述文本：本次构建就能使用更新后的数值。
        ui.ch_id("value").text(format!("Count: {}", counter.value));
    });
}
