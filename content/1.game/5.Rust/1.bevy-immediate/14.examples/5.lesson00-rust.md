struct Counter {
    value: u32,
}

fn increment(counter: &mut Counter) {
    counter.value += 1;
}

trait Describe {
    fn description(&self) -> String;
}

impl Describe for Counter {
    fn description(&self) -> String {
        format!("Count: {}", self.value)
    }
}

fn main() {
    let mut counter = Counter { value: 0 };
    increment(&mut counter);
    let add_two = |value: u32| value + 2;
    counter.value = add_two(counter.value);
    println!("{}", counter.description());
    let mut names = vec![String::from("Alice"), String::from("Bob")];
    names.retain(|name| name != "Alice");
    println!("Remaining: {}", names[0]);
}
