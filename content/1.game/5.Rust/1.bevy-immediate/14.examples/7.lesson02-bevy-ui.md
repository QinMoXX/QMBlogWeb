use bevy::prelude::*;
use bevy_immediate_tutorial::{app, camera, panel};

#[derive(Component)]
struct CounterLabel;

#[derive(Resource, Default)]
struct Counter {
    value: u32,
}

fn main() {
    app("02 - Bevy UI")
        .init_resource::<Counter>()
        .add_systems(Startup, (camera, setup))
        .add_systems(Update, (read_keyboard, update_label).chain())
        .run();
}

fn setup(mut commands: Commands) {
    commands.spawn(panel()).with_children(|parent| {
        parent.spawn((Text::new("Press Space to add one"), TextFont::default()));
        parent.spawn((CounterLabel, Text::new("Count: 0"), TextFont::default()));
    });
}

fn read_keyboard(keys: Res<ButtonInput<KeyCode>>, mut counter: ResMut<Counter>) {
    if keys.just_pressed(KeyCode::Space) {
        counter.value += 1;
    }
}

fn update_label(counter: Res<Counter>, mut labels: Query<&mut Text, With<CounterLabel>>) {
    for mut text in &mut labels {
        let next = format!("Count: {}", counter.value);
        if text.0 != next {
            text.0 = next;
        }
    }
}
