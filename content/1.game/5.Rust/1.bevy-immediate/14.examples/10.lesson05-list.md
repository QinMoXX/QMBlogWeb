use bevy::prelude::*;
use bevy_immediate::{
    BevyImmediatePlugin, ImmCtx,
    ui::{CapsUi, clicked::ImmUiClicked, text::ImmUiText},
};
use bevy_immediate_tutorial::{app, button, camera, panel};

struct Item {
    id: u64,
    title: String,
}

#[derive(Resource, Default)]
struct ListState {
    next_id: u64,
    items: Vec<Item>,
    show_help: bool,
}

fn main() {
    app("05 - Dynamic list")
        .init_resource::<ListState>()
        .add_plugins(BevyImmediatePlugin::<CapsUi>::new())
        .add_systems(Startup, camera)
        .add_systems(Update, draw_ui)
        .run();
}

fn draw_ui(ctx: ImmCtx<CapsUi>, mut state: ResMut<ListState>) {
    let mut ui = ctx.build_immediate_root("lesson05");
    ui.ch_id("panel").on_spawn_insert(panel).add(|ui| {
        let mut add = ui.ch_id("add").on_spawn_insert(button).add(|ui| {
            ui.ch_id("caption").text("Add item");
        });
        if add.clicked() {
            let id = state.next_id;
            state.next_id += 1;
            state.items.push(Item {
                id,
                title: format!("Item {id} (click to remove)"),
            });
        }

        let mut reverse = ui.ch_id("reverse").on_spawn_insert(button).add(|ui| {
            ui.ch_id("caption").text("Reverse order");
        });
        if reverse.clicked() {
            state.items.reverse();
        }

        let mut help = ui.ch_id("help-button").on_spawn_insert(button).add(|ui| {
            ui.ch_id("caption").text("Toggle help");
        });
        if help.clicked() {
            state.show_help = !state.show_help;
        }
        if state.show_help {
            ui.ch_id("help-text")
                .text("Add items, reorder them, or click one to remove it.");
        }

        let mut remove_id = None;
        for item in &state.items {
            let mut row = ui
                .ch_id(("item", item.id))
                .on_spawn_insert(button)
                .add(|ui| {
                    ui.ch_id("caption").text(item.title.as_str());
                });
            if row.clicked() {
                remove_id = Some(item.id);
            }
        }
        // 遍历时只读 Vec，遍历结束后再删除，避免借用冲突。
        if let Some(id) = remove_id {
            state.items.retain(|item| item.id != id);
        }
    });
}
