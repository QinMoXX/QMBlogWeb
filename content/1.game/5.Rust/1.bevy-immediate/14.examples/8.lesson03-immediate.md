```rust
use bevy::prelude::*;
use bevy_immediate::{
    BevyImmediatePlugin, ImmCtx,
    ui::{CapsUi, text::ImmUiText},
};
use bevy_immediate_tutorial::{app, camera, panel};

fn main() {
    app("03 - Immediate UI")
        .add_plugins(BevyImmediatePlugin::<CapsUi>::new())
        .add_systems(Startup, camera)
        .add_systems(Update, draw_ui)
        .run();
}

fn draw_ui(ctx: ImmCtx<CapsUi>) {
    let mut ui = ctx.build_immediate_root("lesson03");
    ui.ch_id("panel").on_spawn_insert(panel).add(|ui| {
        ui.ch_id("title").text("Hello, bevy_immediate!");
        ui.ch_id("description").text("This code runs every frame.");
    });
}
```
