use bevy::prelude::*;
use bevy_immediate::{
    BevyImmediatePlugin, CapSet, ImmCapAccessRequests, ImmCapability, ImmCtx, ImmEntity, ImplCap,
    ImplCapsEmpty, impl_capability_set,
    ui::{
        base::CapabilityUiBase,
        layout_order::CapabilityUiLayoutOrder,
        text::{CapabilityUiText, ImmUiText},
    },
};
use bevy_immediate_tutorial::{app, camera, panel};

pub struct CapabilityHighlight;

impl ImmCapability for CapabilityHighlight {
    fn build<Caps: CapSet>(app: &mut App, requests: &mut ImmCapAccessRequests<Caps>) {
        requests.request_component_write::<BackgroundColor>(app.world_mut());
    }
}

trait ImmHighlight {
    fn highlighted(self, enabled: bool) -> Self;
}

impl<Caps: ImplCap<CapabilityHighlight>> ImmHighlight for ImmEntity<'_, '_, '_, Caps> {
    fn highlighted(mut self, enabled: bool) -> Self {
        let desired = if enabled {
            Color::srgb(0.65, 0.28, 0.08)
        } else {
            Color::srgb(0.12, 0.16, 0.22)
        };
        if let Ok(Some(mut current)) = self.cap_get_component_mut::<BackgroundColor>() {
            if current.0 != desired {
                current.0 = desired;
            }
            return self;
        }
        // 新实体的组件可能尚未可查询，因此排队插入。
        self.entity_commands().insert(BackgroundColor(desired));
        self
    }
}

pub struct TutorialCaps;
impl_capability_set!(
    TutorialCaps,
    ImplTutorialCaps > ImplCapsEmpty,
    (
        CapabilityUiBase,
        CapabilityUiLayoutOrder,
        CapabilityUiText,
        CapabilityHighlight
    )
);

#[derive(Resource, Default)]
struct HighlightState(bool);

fn main() {
    app("08 - Custom capability")
        .init_resource::<HighlightState>()
        .add_plugins(BevyImmediatePlugin::<TutorialCaps>::new())
        .add_systems(Startup, camera)
        .add_systems(Update, draw_ui)
        .run();
}

fn draw_ui(
    ctx: ImmCtx<TutorialCaps>,
    keys: Res<ButtonInput<KeyCode>>,
    mut state: ResMut<HighlightState>,
) {
    if keys.just_pressed(KeyCode::Space) {
        state.0 = !state.0;
    }
    let mut ui = ctx.build_immediate_root("lesson08");
    ui.ch_id("panel").on_spawn_insert(panel).add(|ui| {
        ui.ch_id("help").text("Press Space to toggle highlight");
        ui.ch_id("label")
            .text("A custom capability")
            .highlighted(state.0);
    });
}
