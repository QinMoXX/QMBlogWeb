//! 纯 Rust 的教学模型：没有窗口，也不依赖 Bevy API。
//! 用完整字符串路径充当 ID，用数字充当实体，只展示复用和清理。
use std::collections::{HashMap, HashSet};

#[derive(Default)]
struct MiniUi {
    next_entity: u64,
    entities: HashMap<String, u64>,
    visited: HashSet<String>,
}

impl MiniUi {
    fn begin_frame(&mut self) {
        self.visited.clear();
    }

    fn node(&mut self, path: &str) -> u64 {
        assert!(
            self.visited.insert(path.to_owned()),
            "duplicate id in one frame: {path}"
        );
        if let Some(&entity) = self.entities.get(path) {
            println!("reuse  {path} -> entity {entity}");
            return entity;
        }
        let entity = self.next_entity;
        self.next_entity += 1;
        self.entities.insert(path.to_owned(), entity);
        println!("spawn  {path} -> entity {entity}");
        entity
    }

    fn end_frame(&mut self) {
        // 先收集再排序，使控制台输出顺序稳定，便于对照教程。
        let mut removed: Vec<String> = self
            .entities
            .keys()
            .filter(|path| !self.visited.contains(*path))
            .cloned()
            .collect();
        removed.sort();
        for path in removed {
            let entity = self.entities.remove(&path).unwrap();
            println!("remove {path} -> entity {entity}");
        }
    }
}

fn main() {
    let mut ui = MiniUi::default();
    for (frame, show_help) in [(1, true), (2, true), (3, false)] {
        println!("--- frame {frame} ---");
        ui.begin_frame();
        ui.node("root/title");
        if show_help {
            ui.node("root/help");
        }
        ui.end_frame();
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn identity_survives_reordering_but_not_removal() {
        let mut ui = MiniUi::default();
        ui.begin_frame();
        let first_a = ui.node("root/a");
        let first_b = ui.node("root/b");
        ui.end_frame();

        ui.begin_frame();
        assert_eq!(ui.node("root/b"), first_b);
        assert_eq!(ui.node("root/a"), first_a);
        ui.end_frame();

        ui.begin_frame();
        ui.node("root/b");
        ui.end_frame();
        assert!(!ui.entities.contains_key("root/a"));

        ui.begin_frame();
        assert_ne!(ui.node("root/a"), first_a);
        ui.end_frame();
    }

    #[test]
    #[should_panic(expected = "duplicate id")]
    fn repeated_sibling_id_is_rejected() {
        let mut ui = MiniUi::default();
        ui.begin_frame();
        ui.node("root/a");
        ui.node("root/a");
    }
}
